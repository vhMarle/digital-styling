function handleFileSelect(e) { if (e.target.files.length > 0) { const file = e.target.files[0]; document.getElementById('drop-text').innerHTML = `✅ Selected: ${file.name}`; // Auto-extract metadata from filename or default values const fname = file.name.toLowerCase(); document.getElementById('item-name').value = file.name.replace(/\.[^/.]+$/, "").replace(/[-_]/g, ' '); document.getElementById('item-category').value = fname.includes('shoe') ? 'Shoes' : fname.includes('jean') || fname.includes('pant') ? 'Bottoms' : 'Dresses'; populateSubTypes(); document.getElementById('item-brand').value = 'Lulus'; document.getElementById('item-color').value = fname.includes('olive') ? 'Olive Green' : fname.includes('green') ? 'Dark Green' : 'Black'; document.getElementById('item-size').value = 'US 8'; const reader = new FileReader(); reader.onload = (event) => processImageSegmentation(event.target.result); reader.readAsDataURL(file); } }
