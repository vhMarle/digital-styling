function handleFileSelect(e) { if (e.target.files && e.target.files.length > 0) { const file = e.target.files[0]; processSelectedFile(file); } } function processSelectedFile(file) { document.getElementById('drop-text').innerHTML = `✅ Selected: ${file.name}`; // Auto-populate item metadata from filename const fname = file.name.toLowerCase(); const cleanName = file.name.replace(/\.[^/.]+$/, "").replace(/[-_]/g, ' '); document.getElementById('item-name').value = cleanName.charAt(0).toUpperCase() + cleanName.slice(1); const isShoe = fname.includes('shoe') || fname.includes('boot') || fname.includes('heel'); const isBottom = fname.includes('jean') || fname.includes('pant') || fname.includes('skirt'); document.getElementById('item-category').value = isShoe ? 'Shoes' : isBottom ? 'Bottoms' : 'Dresses'; populateSubTypes(); document.getElementById('item-brand').value = fname.includes('lulus') ? 'Lulus' : 'ASOS Luxe'; document.getElementById('item-color').value = fname.includes('olive') ? 'Olive Green' : fname.includes('green') ? 'Dark Green' : fname.includes('black') ? 'Black' : 'Dark Green'; document.getElementById('item-size').value = 'US 8'; const reader = new FileReader(); reader.onload = (event) => processImageSegmentation(event.target.result); reader.readAsDataURL(file); }
